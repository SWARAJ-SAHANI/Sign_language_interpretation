from flask import Flask, request, jsonify
from threading import Timer

app = Flask(__name__)

_say_timer = None
_last_text = ""

def _fire_say():
    say(_last_text)

@app.route('/speak', methods=['POST'])
def speak():
    global _say_timer, _last_text

    data = request.get_json(silent=True) or {}
    text = data.get('text', '').strip()
    if not text:
        return jsonify({"status": "no text"}), 400

    # 1) always call dummy immediately
    dummy()

    # 2) cancel & replace the pending timer
    if _say_timer:
        _say_timer.cancel()

    _last_text = text
    _say_timer = Timer(1.0, _fire_say)
    _say_timer.daemon = True
    _say_timer.start()

    return jsonify({"status": "queued", "text": text})
